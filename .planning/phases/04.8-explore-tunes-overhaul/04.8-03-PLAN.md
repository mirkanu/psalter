```js
const { chromium } = require('/usr/lib/node_modules/playwright');
async function main() {
  const browser = await chromium.launch({
    executablePath: '/tmp/pw-browsers/chromium-1217/chrome-linux/chrome',
    args: ['--no-sandbox'],
  });
  const page = await browser.newPage();
  const results = [];

  // SC-1: /explore has "When you..." section
  await page.goto('http://localhost:3005/explore', { waitUntil: 'networkidle' });
  const exploreBody = await page.textContent('body');
  results.push(['SC-2: /explore has "When you..." section', exploreBody.includes('When you...')]);
  results.push(['SC-2: /explore has "Topics" section', exploreBody.includes('Topics')]);
  results.push(['SC-2: /explore has "Nave\'s Topics" section', exploreBody.includes("Nave's Topics")]);
  results.push(['SC-2: /explore has "Messianic Psalms" section', exploreBody.includes('Messianic Psalms')]);

  // SC-3: /tunes has mood filter and expanded cards
  await page.goto('http://localhost:3005/tunes', { waitUntil: 'networkidle' });
  const tunesBody = await page.textContent('body');
  results.push(['SC-3: /tunes has "All Moods" filter', tunesBody.includes('All Moods')]);
  results.push(['SC-3: /tunes has "All Meters" filter', tunesBody.includes('All Meters')]);
  const tuneCardCount = await page.locator('a[href^="/tunes/"]').count();
  results.push(['SC-3: /tunes has at least 1 tune card', tuneCardCount > 0]);

  // SC-4: /search redirect — follow redirect and check final URL
  const searchResp = await page.goto('http://localhost:3005/search', { waitUntil: 'networkidle' });
  const finalUrl = page.url();
  results.push(['SC-4: /search redirects to /psalms', finalUrl.includes('/psalms')]);

  // SC-4: SiteHeader has no Search nav link
  await page.goto('http://localhost:3005/', { waitUntil: 'networkidle' });
  const headerHtml = await page.locator('header').innerHTML();
  results.push(['SC-4: SiteHeader has no Search link', !headerHtml.includes('href="/search"')]);
  results.push(['SC-4: SiteHeader has Psalms link', headerHtml.includes('/psalms')]);
  results.push(['SC-4: SiteHeader has Tunes link', headerHtml.includes('/tunes')]);
  results.push(['SC-4: SiteHeader has Explore link', headerHtml.includes('/explore')]);

  results.forEach(([label, ok]) => console.log(`${ok ? 'PASS' : 'FAIL'}: ${label}`));
  await browser.close();
  if (results.some(([, ok]) => !ok)) process.exit(1);
}
main().catch(e => { console.error(e); process.exit(1); });
```
