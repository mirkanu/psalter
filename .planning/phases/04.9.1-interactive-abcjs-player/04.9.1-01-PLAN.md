Create `src/components/AbcPlayer.tsx` ('use client'). Props:

```ts
interface AbcPlayerProps {
  abc: string                          // soprano-only ABC notation (from tunes.abc_notation)
  title?: string
  staffJpgUrl?: string | null          // tunes.score_jpg_url
  solfegeJpgUrl?: string | null        // tunes.solfege_jpg_url (for the show-original toggle on the psalm page where solfège may be the active scan)
  tuneName?: string
  initialMode?: 'staff' | 'solfege'    // which JPEG to use when "Show original" is on; defaults to 'staff'
}
```

Internal state:
- `transpose: number` — default 0, range -6..+6
- `bpm: number` — default 100 (typical psalm-tune tempo); range 40..200, step 5
- `showOriginal: boolean` — default false; when true, render the JPEG instead of the SVG
- `isPlaying: boolean`
- `audioReady: boolean` — set true after first successful `synth.init + prime`
- `audioError: string | null`

Refs:
- `containerRef: RefObject<HTMLDivElement>` — abcjs SVG mount point
- `visualObjRef: MutableRefObject<any | null>` — current visualObj
- `synthRef: MutableRefObject<any | null>` — current CreateSynth instance
- `timingRef: MutableRefObject<any | null>` — current TimingCallbacks
- `lastHighlightedRef: MutableRefObject<SVGElement[] | null>` — previously highlighted SVG nodes (so we can un-highlight before re-applying)

Behaviour:

1. **Render effect** — runs on mount and whenever `abc`, `transpose`, or `bpm` changes (but NOT `isPlaying` or `showOriginal`):
   - If `showOriginal` is true OR no container ref, skip (the JSX shows the `<img>` instead).
   - Stop and null out any prior synth/timingCallbacks; reset `isPlaying` to false.
   - Clear `containerRef.current.innerHTML`.
   - Call `abcjs.renderAbc(container, abc, { responsive: 'resize', add_classes: true, visualTranspose: transpose, defaultTempo: { duration: 0.25, bpm } })` — wrap in try/catch and set `audioError` on failure.
   - Store `visualObj = visualObjs[0]` in `visualObjRef`.

2. **Play handler** (`onPlay`):
   - If `synthRef.current` is null OR a re-render has occurred since last init: create a fresh `abcjs.synth.CreateSynth()`, call `await synth.init({ visualObj: visualObjRef.current, options: { soundFontUrl: 'https://paulrosen.github.io/midi-js-soundfonts/abcjs/' }, millisecondsPerMeasure: visualObjRef.current.millisecondsPerMeasure(bpm) })`, then `await synth.prime()`. Store in `synthRef`. Set `audioReady = true`. On failure: set `audioError = 'Audio not available in this browser.'` and bail.
   - Create a fresh `abcjs.TimingCallbacks(visualObjRef.current, { eventCallback: highlightEvent })`. Store in `timingRef`.
   - Call `synth.start()` and `timingRef.current.start()`. Set `isPlaying = true`.

3. **Pause handler** (`onPause`):
   - `synthRef.current?.pause()` and `timingRef.current?.stop()`. Set `isPlaying = false`. Clear last highlight.

4. **highlightEvent(ev)** — eventCallback body:
   - If previous highlighted SVG nodes exist in `lastHighlightedRef.current`, remove the `abcjs-current-note` class from each.
   - If `!ev || !ev.elements`: stop (end of tune) — call `onPause()` internally and return.
   - For each element-group in `ev.elements`, for each SVG node: add class `abcjs-current-note`. Store the flat list in `lastHighlightedRef.current`.

5. **CSS** — at top of `AbcPlayer.tsx` add a `<style jsx global>` (or inline `<style>`) block:
   ```css
   .abcjs-current-note { fill: #2563eb !important; }       /* tailwind blue-600 */
   .abcjs-current-note path { fill: #2563eb !important; }
   ```

6. **Cleanup on unmount**: stop synth + timingCallbacks; clear container.

JSX layout (Tailwind, mobile-first; shadcn components):

```tsx

  {/* Notation area: SVG OR original JPEG */}

    <div className="relative w-full aspect-[3/2]">
      <img

      disabled={!staffJpgUrl && !solfegeJpgUrl}

      {showOriginal ? 'Show notation' : 'Show original'}
```

```ts
import { test, expect } from '@playwright/test'

const BASE = process.env.TEST_BASE_URL ?? 'https://psalter.gsdlabs.dev'

test.describe('AbcPlayer on /tunes/[id]', () => {
  // Pick the first tune that renders an ABC player. Strategy: hit /tunes,
  // open the first card, then assert the player is present.
  let tuneUrl: string

  test.beforeAll(async ({ browser }) => {
    const ctx = await browser.newContext()
    const page = await ctx.newPage()
    await page.goto(`${BASE}/tunes`)
    // Find first tune link that, when opened, has an [data-testid="abc-play-button"]
    const links = await page.locator('a[href^="/tunes/"]').elementHandles()
    for (const a of links.slice(0, 20)) {
      const href = await a.getAttribute('href')
      if (!href) continue
      const probe = await ctx.newPage()
      await probe.goto(`${BASE}${href}`)
      if (await probe.locator('[data-testid="abc-play-button"]').count()) {
        tuneUrl = `${BASE}${href}`
        await probe.close()
        break
      }
      await probe.close()
    }
    await ctx.close()
    if (!tuneUrl) throw new Error('No tune with AbcPlayer found in first 20 tunes')
  })

  test('PLAYER-01: Play triggers audio and highlights a note', async ({ page }) => {
    await page.goto(tuneUrl, { waitUntil: 'networkidle' })
    const playBtn = page.getByTestId('abc-play-button')
    await expect(playBtn).toBeVisible()

    // Click Play — button label flips to "Pause"
    await playBtn.click()
    await expect(playBtn).toContainText(/Pause/i, { timeout: 5000 })

    // Within a few seconds an SVG note has the highlight class
    const highlighted = page.locator('.abcjs-current-note')
    await expect(highlighted.first()).toBeVisible({ timeout: 5000 })
  })

  test('PLAYER-02: Transpose changes the SVG render', async ({ page }) => {
    await page.goto(tuneUrl, { waitUntil: 'networkidle' })
    const svgBefore = await page.locator('svg.abcjs').first().innerHTML().catch(async () =>
      await page.locator('svg').first().innerHTML()
    )

    // Open shadcn Select trigger
    await page.getByTestId('abc-transpose-select').click()
    await page.getByRole('option', { name: '+3' }).click()

    // After re-render, SVG markup differs
    await page.waitForTimeout(500)
    const svgAfter = await page.locator('svg').first().innerHTML()
    expect(svgAfter).not.toEqual(svgBefore)
  })

  test('PLAYER-03: BPM controls change the displayed value', async ({ page }) => {
    await page.goto(tuneUrl, { waitUntil: 'networkidle' })
    const bpmValue = page.getByTestId('abc-bpm-value')
    const initial = Number(await bpmValue.textContent())
    await page.getByRole('button', { name: 'Increase tempo' }).click()
    await page.getByRole('button', { name: 'Increase tempo' }).click()
    const after = Number(await bpmValue.textContent())
    expect(after).toBe(initial + 10)
  })

  test('PLAYER-04: Show original toggles to JPEG', async ({ page }) => {
    await page.goto(tuneUrl, { waitUntil: 'networkidle' })
    const toggle = page.getByTestId('abc-show-original-toggle')

    // Pre-toggle: SVG present
    await expect(page.locator('svg').first()).toBeVisible()

    await toggle.click()
    // Post-toggle: <img> present, SVG gone (or no longer visible)
    await expect(page.locator('img[alt*="Original score"]')).toBeVisible()

    await toggle.click()
    // Returns to SVG
    await expect(page.locator('svg').first()).toBeVisible()
  })

  test('PLAYER-05: Controls remain reachable at 375px', async ({ page }) => {

    // No horizontal overflow on the controls row

  })
})
```
