       ```sql
       SELECT id, name, LENGTH(abc_notation) AS abc_len

       LIMIT 20;
       ```

       ```sql
       SELECT id, name, LENGTH(abc_notation) AS abc_len
       FROM tunes
       WHERE meter = 'CM' AND abc_notation IS NOT NULL

       LIMIT 10;
       ```

       ```sql
       SELECT id, name, RIGHT(abc_notation, 60) AS tail
       FROM tunes
       WHERE meter = 'CM' AND abc_notation IS NOT NULL
       ORDER BY name
       LIMIT 30;
       ```

       ```sql
       SELECT psv.psalm_id, psv.version_label, psv.meter AS lyric_meter,

       FROM psalm_version_tunes pvt
       JOIN psalm_versions psv ON pvt.psalm_version_id = psv.id
       JOIN tunes t ON pvt.tune_id = t.id
       WHERE t.name = 'Aurelia';
       ```
