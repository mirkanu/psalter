Slugify function (verbatim from UI-SPEC — use exactly this implementation):
```typescript
function slugify(name: string): string {
  return name
    .toLowerCase()
    .replace(/['']/g, '')
    .replace(/\s+/g, '-')
    .replace(/[^a-z0-9-]/g, '')
    .replace(/-+/g, '-')
    .replace(/^-+|-+$/g, '')  // trim leading/trailing dashes
}
```

Slug collision handling for Nave's topics (RESEARCH.md verified — only 1 collision):
```typescript
// Build slug map to detect collisions
function buildNavesSlugMap(topics: Array<{ id: number; name: string }>): Map<number, string> {
  const slugCount = new Map<string, number>()
  for (const t of topics) {
    const base = slugify(t.name)
    slugCount.set(base, (slugCount.get(base) ?? 0) + 1)
  }
  const result = new Map<number, string>()
  for (const t of topics) {
    const base = slugify(t.name)
    result.set(t.id, (slugCount.get(base) ?? 1) > 1 ? `${base}-${t.id}` : base)
  }
  return result
}
// "Mercy Seat" (id 269) → "mercy-seat-269"
// "Mercy-Seat" (id 477) → "mercy-seat-477"
// All other topics use plain slug
```
