```typescript
export function parseLyrics(raw: string, meter: string | null | undefined): ParseResult {
  const normMeter = normaliseMeter(meter)
  const linesPerStanza = normMeter ? METRICAL_LINES[normMeter] : undefined
  if (!linesPerStanza) {
    // Fallback: many CM rows have meter strings like "CM (common meter, 86 86)" — try a fuzzy match
    for (const key of Object.keys(METRICAL_LINES)) {
      if (normMeter && normMeter.toUpperCase().includes(key)) {
        return continueWithMeter(raw, METRICAL_LINES[key])
      }
    }
    return { ok: false, reason: `unknown meter: ${meter} (not in METRICAL_LINES)` }
  }
  return continueWithMeter(raw, linesPerStanza)
}

function continueWithMeter(raw: string, linesPerStanza: number): ParseResult {
  // Detect F-5 colophon: any line starts with '#'
  if (/^#/m.test(raw)) {
    const lineIdx = raw.split('\n').findIndex((l) => l.startsWith('#'))
    return { ok: false, reason: 'colophon line detected (F-5)', line: lineIdx }
  }

  // Step 1+2: flatten all non-blank lines preserving order
  const allLines: { raw: string; verseRef?: number }[] = []
  for (const blobGroup of raw.split(/\n\s*\n/)) {
    for (const line of blobGroup.split(/\n/)) {
      const trimmed = line.replace(/[ \t]+$/, '')
      if (!trimmed) continue
      // F-2/F-3: leading digit run with optional whitespace
      const m = trimmed.match(/^(\d+)\s*(.*)$/)
      if (m && m[2].length > 0) {
        allLines.push({ raw: m[2], verseRef: parseInt(m[1], 10) })
      } else {
        allLines.push({ raw: trimmed })
      }
    }
  }

  // Step 3: chunk into stanzas of linesPerStanza
  if (allLines.length % linesPerStanza !== 0) {
    return {
      ok: false,
      reason: `line count ${allLines.length} not a multiple of ${linesPerStanza} (meter F-6)`,
    }
  }

  const stanzas: StructuredLyrics = []
  for (let i = 0; i < allLines.length; i += linesPerStanza) {
    const lines: Line[] = allLines.slice(i, i + linesPerStanza).map((l) => {
      const out: Line = { text: l.raw }
      if (l.verseRef !== undefined) out.bibleVerseRef = l.verseRef
      return out
    })
    stanzas.push({ index: stanzas.length, lines })
  }

  return { ok: true, stanzas }
}
```

export function serialiseLyrics(stanzas: StructuredLyrics): string {
  return stanzas
    .map((stanza) =>
      stanza.lines
        .map((line) =>
          line.bibleVerseRef !== undefined
            ? `${line.bibleVerseRef}${line.text}`
            : line.text,
        )
        .join('\n'),
    )
    .join('\n\n')
}
