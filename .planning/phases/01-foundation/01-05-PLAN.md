**Step 1 — Write scripts/verify-migration.ts:**

```typescript
#!/usr/bin/env npx tsx
/**
 * Migration verification script.
 * Run: npx tsx scripts/verify-migration.ts
 * Run (quick mode — row counts only): npx tsx scripts/verify-migration.ts --quick
 *
 * Covers:
 *   MIGR-01: All tables present with correct row counts
 *   MIGR-02: Tune JPG files present on local filesystem (TUNES_DIR)
 *   MIGR-03: Spot-check records match expected Airtable values
 *   MIGR-04: Idempotency — run twice and assert same counts
 */

import 'dotenv/config'
import { drizzle } from 'drizzle-orm/postgres-js'
import postgres from 'postgres'
import { count, eq, isNull } from 'drizzle-orm'
import * as schema from '../src/db/schema'

const client = postgres(process.env.DATABASE_URL!)
const db = drizzle({ client, schema })

const args = process.argv.slice(2)
const QUICK_MODE = args.includes('--quick')

let failures = 0

function assert(condition: boolean, message: string): void {
  if (condition) {
    console.log(`  ✓ ${message}`)
  } else {
    console.error(`  ✗ FAIL: ${message}`)
    failures++
  }
}

// ─── Row Count Checks ─────────────────────────────────────────────────────────

async function checkRowCounts(): Promise<Record<string, number>> {
  console.log('\n── Row Counts ──')

  const counts: Record<string, number> = {}

  const tables = [
    { name: 'psalms', table: schema.psalms, expected: 150 },
    { name: 'tunes', table: schema.tunes, expected: null },      // exact count TBD after first run
    { name: 'psalm_versions', table: schema.psalmVersions, expected: null },
    { name: 'verses', table: schema.verses, expected: null },
    { name: 'daily_readings', table: schema.dailyReadings, expected: null },
    { name: 'events', table: schema.events, expected: 29 },
    { name: 'service_items', table: schema.serviceItems, expected: null },
    { name: 'messianic_psalms', table: schema.messianicPsalms, expected: 18 },
    { name: 'section_headings', table: schema.sectionHeadings, expected: null },
    { name: 'topics', table: schema.topics, expected: 88 },
    { name: 'naves_topics', table: schema.navesTopics, expected: null },
    { name: 'moods', table: schema.moods, expected: null },
    { name: 'doctrines', table: schema.doctrines, expected: null },
  ]

  for (const { name, table, expected } of tables) {
    const [{ value }] = await db.select({ value: count() }).from(table)
    const n = Number(value)
    counts[name] = n

    if (expected !== null) {
      assert(n === expected, `${name}: ${n} rows (expected ${expected})`)
    } else {
      assert(n > 0, `${name}: ${n} rows (expected >0)`)
      console.log(`    [info] ${name}: ${n} rows — update EXPECTED after first run`)
    }
  }

  // Junction tables
  const junctions = [
    { name: 'psalm_topics', table: schema.psalmTopics },
    { name: 'psalm_version_tunes', table: schema.psalmVersionTunes },
    { name: 'tune_moods', table: schema.tuneMoods },
    { name: 'verse_naves_topics', table: schema.verseNavesTopics },
    { name: 'verse_doctrines', table: schema.verseDoctrines },
  ]

  for (const { name, table } of junctions) {
    const [{ value }] = await db.select({ value: count() }).from(table)
    const n = Number(value)
    counts[name] = n
    assert(n > 0, `${name}: ${n} junction rows (expected >0)`)
  }

  return counts
}

// ─── Local Image Checks ─────────────────────────────────────────────────────────

async function checkLocalImages(): Promise<void> {
  console.log('\n── Local Image Checks ──')

  // Check no Airtable URLs stored
  const airtableUrls = await db.select({ url: schema.tunes.scoreJpgUrl })
    .from(schema.tunes)
    .where(eq(schema.tunes.scoreJpgUrl, 'airtableusercontent.com'))
  const allPaths = await db.select({ url: schema.tunes.scoreJpgUrl })
    .from(schema.tunes)

  const airtableStored = allPaths.filter(r => r.url?.includes('airtableusercontent.com'))
  assert(airtableStored.length === 0, `No Airtable temporary URLs stored in database (found ${airtableStored.length})`)

  const withPath = allPaths.filter(r => r.url)
  assert(withPath.length > 0, `At least 1 tune has a score_jpg_url (found ${withPath.length})`)

  if (QUICK_MODE) {
    console.log('  [quick mode] Skipping filesystem checks')
    return
  }

  // Check files exist on local filesystem
  const { existsSync } = await import('node:fs')
  const TUNES_DIR = process.env.TUNES_DIR ?? '/app/public/tunes'
  const sample = withPath.slice(0, 5)
  for (const { url } of sample) {
    if (!url) continue
    const filename = url.replace('/tunes/', '')
    const filepath = `${TUNES_DIR}/${filename}`
    assert(existsSync(filepath), `Tune image file exists on disk: ${filename}`)
  }
}

// ─── Spot-Check Records ───────────────────────────────────────────────────────

async function checkSpotRecords(): Promise<void> {
  if (QUICK_MODE) {
    console.log('\n── Spot Checks ── [skipped in quick mode]')
    return
  }

  console.log('\n── Spot Checks ──')

  // Psalm 23 — "The LORD is my shepherd"
  const psalm23 = await db.query.psalms.findFirst({ where: eq(schema.psalms.id, 23) })
  assert(psalm23 !== undefined, 'Psalm 23 exists in database')
  assert(psalm23?.kjvText?.includes('LORD is my shepherd') ?? false, 'Psalm 23 KJV text contains "LORD is my shepherd"')

  // Psalm 1 — "Blessed is the man"
  const psalm1 = await db.query.psalms.findFirst({ where: eq(schema.psalms.id, 1) })
  assert(psalm1 !== undefined, 'Psalm 1 exists in database')
  assert(psalm1?.kjvText?.includes('Blessed is the man') ?? false, 'Psalm 1 KJV text contains "Blessed is the man"')

  // Psalm 119 — longest psalm, should have many verses
  const psalm119Verses = await db.select({ value: count() })
    .from(schema.verses)
    .where(eq(schema.verses.psalmId, 119))
  const v119count = Number(psalm119Verses[0].value)
  assert(v119count > 100, `Psalm 119 has >100 verses (found ${v119count} — expected ~176)`)

  // Events — should have some service items linked
  const eventsWithItems = await db.select({ value: count() })
    .from(schema.serviceItems)
  assert(Number(eventsWithItems[0].value) > 0, 'Service items table is not empty')

  // Topics — psalms should be tagged
  const psalmTopicCount = await db.select({ value: count() }).from(schema.psalmTopics)
  assert(Number(psalmTopicCount[0].value) > 0, 'Psalm-topic links exist')

  // Tunes — meter normalised (no long strings)
  const longMeterTunes = await db.select({ meter: schema.tunes.meter })
    .from(schema.tunes)
    .where(eq(schema.tunes.meter, 'Common Meter (CM, 86 86)'))
  assert(longMeterTunes.length === 0, 'No un-normalised meter values (e.g., "Common Meter (CM, 86 86)") stored')

  // Daily readings — should have ~365
  const drCount = await db.select({ value: count() }).from(schema.dailyReadings)
  assert(Number(drCount[0].value) >= 365, `Daily readings >= 365 (found ${drCount[0].value})`)
}

// ─── Idempotency Check (MIGR-04) ─────────────────────────────────────────────

async function checkIdempotency(firstRunCounts: Record<string, number>): Promise<void> {
  if (QUICK_MODE) {
    console.log('\n── Idempotency Check ── [skipped in quick mode]')
    return
  }

  console.log('\n── Idempotency Check (MIGR-04) ──')
  console.log('  Re-running migration to test delta idempotency...')

  // Import and re-run migration
  // We do this via child_process to isolate the run
  const { execSync } = await import('child_process')
  try {
    execSync('npx tsx scripts/migrate-airtable.ts', {
      stdio: 'pipe',
      cwd: process.cwd(),
      env: process.env,
    })
    console.log('  Second migration run completed')
  } catch (err) {
    assert(false, `Second migration run failed: ${err}`)
    return
  }

  // Check counts match
  const tables = [
    { name: 'psalms', table: schema.psalms },
    { name: 'tunes', table: schema.tunes },
    { name: 'verses', table: schema.verses },
    { name: 'events', table: schema.events },
  ]

  for (const { name, table } of tables) {
    const [{ value }] = await db.select({ value: count() }).from(table)
    const n = Number(value)
    const expected = firstRunCounts[name]
    assert(n === expected, `${name}: count unchanged after re-run (${expected} → ${n})`)
  }
}

// ─── Main ─────────────────────────────────────────────────────────────────────

async function main() {
  console.log('=== Migration Verification ===')
  if (QUICK_MODE) console.log('[QUICK MODE — row counts only]')

  const counts = await checkRowCounts()
  await checkLocalImages()
  await checkSpotRecords()
  await checkIdempotency(counts)

  console.log('\n═══════════════════════════════')
  if (failures === 0) {
    console.log('✓ All checks passed')
  } else {
    console.error(`✗ ${failures} check(s) failed`)
  }
  console.log('═══════════════════════════════')

  await client.end()
  process.exit(failures > 0 ? 1 : 0)
}

main().catch((err) => {
  console.error('Verification failed:', err)
  client.end()
  process.exit(1)
})
```
