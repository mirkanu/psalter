<!-- Sol-fa half-section boundary rule (from .continue-here.md):
     The sol-fa string has ONE `||` in the middle (separating two 8-bar halves)
     plus a trailing `||` (Amen marker). parseVoiceLine already strips the Amen
     section (lastIndexOf `||`), so after parsing the soprano produces events for
     the full 32-note (CM) or 32-note (LM) tune body.

     Phrase index to note-event range mapping for CM (32 events total):
       phrase 0: events[0..7]   (8 notes, lines 1 of the CM stanza)
       phrase 1: events[8..13]  (6 notes, line 2)
       phrase 2: events[14..21] (8 notes, line 3)
       phrase 3: events[22..31] (10 notes... or 12 for Crimond — the melisma phrase)

     getSplitPointsForMeter("CM", 4) → [8, 14, 22]
     Phrase boundaries: [0, 8, 14, 22, end]
     So phrase index i → events[boundaries[i]..boundaries[i+1]-1]
-->

<!-- ABC w: field token semantics (from .planning/notes/syllable-alignment-via-abc.md):
     - `_`  = melisma/hold — PREVIOUS syllable continues over this note (CORRECT for passing)
     - `*`  = skip — note gets no syllable at all (WRONG for passing notes)
     - `-`  = word hyphen — splits a word across consecutive notes
     Each token in the w: field aligns positionally to one note in the staff.
     Tokens are space-separated. | in w: re-syncs to the next barline in the music.

    buildWLineFromSolfa(soprano, doh, time, phraseIndex, meter, syllables): string

    Algorithm:
    1. Call getPassingPositions(soprano, doh, time) → boolean[] for all events
    2. Slice to phrase range using getSplitPointsForMeter:
       - boundaries = [0, ...getSplitPointsForMeter(meter, phrasesForMeter(meter))]
       - phraseStart = boundaries[phraseIndex]
       - phraseEnd = boundaries[phraseIndex + 1] ?? totalEvents
       - phrasePassings = passingPositions.slice(phraseStart, phraseEnd)
    3. Count syllabic slots: syllabicCount = phrasePassings.filter(v => !v).length
    4. Split syllables string into tokens via syllabifyForAbc(syllables)
       then split result on /\s+/
    5. If syllabicCount === syllableTokens.length: perfect match
       - Walk phrasePassings: passing position → emit '_', syllabic position → emit next syllable token
    6. If syllabicCount > syllableTokens.length: too many syllabic slots
       - Apply duration heuristic: from the note events in the phrase range,
         find shortest-duration notes at non-passing positions;
         mark shortest notes as passing until syllabicCount === syllableTokens.length
       - Resolve tie-breaks by taking later notes first (finals phrases have run-ins at end)
    7. If syllabicCount < syllableTokens.length: too many syllables (shouldn't happen, log warning)
    8. Join tokens with ' ' and return
    9. Fallback: if getSplitPointsForMeter returns undefined (unknown meter), treat the entire
       passingPositions array as the phrase — no slicing

  ```typescript
  /**
   * Melisma-aware w: line generator for Scottish Psalter tunes.
   *
   * Strategy (hybrid, zero AI cost — see .continue-here.md):
   *   Step 1: Detect passing notes from sol-fa dot-pair syntax (f.r → r is passing)
   *   Step 2: Duration heuristic for residual surplus after dot-pair detection
   *
   * Consumed by NotationRenderer.wLinesForPhrase when solfegeOcrText is available.
   */

  import { getPassingPositions, DOH_SEMITONES, parseSyllable, DEGREE } from './solfege-parser'
  import { getSplitPointsForMeter } from '../scripts/annotate-phrase-breaks'
  import { syllabifyForAbc } from './lyrics'

  // Import phrasesForMeter to find the correct phrase count for the meter
  import { phrasesForMeter } from './abc-phrase-meter-map'

  // Re-export NoteEvent type for consumers
  export type { NoteEvent } from './solfege-parser'

  /**
   * Internal: parse the soprano solfège string into (note, duration, passing) triples
   * for a specific phrase range. Used for duration heuristic.
   */
  function getPhraseNoteEvents(
    soprano: string,
    doh: string,
    time: string,
    phraseStart: number,
    phraseEnd: number,
  ): Array<{ duration: number; passing: boolean }> {
    // We only need duration info; reuse the passing[] array and derive duration
    // from the raw parser. Since parseVoiceLine is not exported, we use a
    // workaround: parse twice — once for passings (already done by caller),
    // once for durations via solFaToAbc.
    //
    // Actually: parseVoiceLine IS accessible via getPassingPositions return value,
    // but we only get booleans. For duration heuristic, parse note durations
    // using a minimal inline version.
    //
    // SIMPLER APPROACH: We know dot-pairs always produce duration=1 (half-beat).
    // Multi-beat notes (duration ≥ 3) are the safest candidates for `_` assignment.
    // We expose this via a separate exported helper.
    //
    // For the heuristic, we need only: which non-passing notes have short duration.
    // Use the getDurationEvents helper below.
    return getDurationEvents(soprano, doh, phraseStart, phraseEnd)
  }

  /**
   * Parse soprano solfège to extract durations for notes in [phraseStart, phraseEnd).
   * Returns duration per event in that range, in order.
   */
  function getDurationEvents(
    soprano: string,
    doh: string,
    phraseStart: number,
    phraseEnd: number,
  ): Array<{ duration: number; passing: boolean }> {
    // Import the full parser internals via the re-exported NoteEvent.
    // Since parseVoiceLine is internal, we call solFaToAbc and parse the
    // duration from the produced ABC string.
    //
    // Simpler: inlining a minimal duration-only parse. The solfège duration
    // tokens are:
    //   dot-pair (. separator) → 1 unit each
    //   single beat (:) → 2 units
    //   hold (—) → extends previous
    //
    // This inline parse only needs durations + passing flag — not pitches.
    const warnings: string[] = []
    let cleaned = soprano.trim()
    const lastDbl = cleaned.lastIndexOf('||')
    if (lastDbl >= 0) cleaned = cleaned.slice(0, lastDbl)
    cleaned = cleaned.replace(/\|\|/g, '|').replace(/\|$/, '').trim()

    const cells = cleaned.split(/\|+/).map((c) => c.trim()).filter(Boolean)
    const allEvents: Array<{ duration: number; passing: boolean }> = []

    for (const cell of cells) {
      const slots = cell.split(':').map((s) => s.trim()).filter(Boolean)
      for (const slot of slots) {
        const rawSubs = slot.split('.')
        const subTokens = rawSubs
          .map((s, i) => {
            let tok = s.trim()
            if (i === 0 && rawSubs.length > 1 && tok.endsWith(',')) tok = tok.slice(0, -1).trim()
            if (i > 0 && tok.startsWith(',')) tok = tok.slice(1).trim()
            return tok
          })
          .filter(Boolean)
        const halfBeat = subTokens.length > 1
        const unitDur = halfBeat ? 1 : 2

        for (let si = 0; si < subTokens.length; si++) {
          const token = subTokens[si]
          const isPassing = halfBeat && si > 0
          if (token === '—' || token === '-') {
            if (allEvents.length > 0) {
              allEvents[allEvents.length - 1].duration += unitDur
            }
          } else {
            allEvents.push({ duration: unitDur, passing: isPassing })
          }
        }
      }
    }

    return allEvents.slice(phraseStart, phraseEnd)
  }

  /**
   * Generate an abcjs w:-field string for one phrase of a melismatic tune.
   *
   * @param soprano       Raw solfège soprano line from DB (solfege_ocr_text soprano field)
   * @param doh           Key letter (e.g. "G", "Bb")
   * @param time          Time signature (e.g. "C", "3/4")
   * @param phraseIndex   0-based phrase index within the tune (0..3 for CM)
   * @param meter         Tune meter string (e.g. "CM", "LM (long meter, 88 88)")
   * @param syllables     Space-separated syllable text for this phrase portion
   *                      (already a single metrical line, e.g. "The Lord's my shep- herd")
   * @param warnings      Mutable array for diagnostic messages (passed in from caller)
   */
  export function buildWLineFromSolfa(
    soprano: string,
    doh: string,
    time: string,
    phraseIndex: number,
    meter: string,
    syllables: string,
    warnings: string[],
  ): string {
    // Step 1: Get passing positions for entire tune
    const allPassings = getPassingPositions(soprano, doh, time)
    const totalEvents = allPassings.length

    // Step 2: Determine phrase note-event range
    const phraseCount = phrasesForMeter(meter)
    const splitPoints = getSplitPointsForMeter(meter, phraseCount)
    let phraseStart = 0
    let phraseEnd = totalEvents

    if (splitPoints) {
      const boundaries = [0, ...splitPoints, totalEvents]
      phraseStart = boundaries[phraseIndex] ?? 0
      phraseEnd = boundaries[phraseIndex + 1] ?? totalEvents
    }

    const phrasePassings = allPassings.slice(phraseStart, phraseEnd)
    const noteCount = phrasePassings.length

    // Step 3: Prepare syllable tokens from the syllabifyForAbc output
    const syllabified = syllabifyForAbc(syllables)
    const sylTokens = syllabified ? syllabified.split(/\s+/).filter(Boolean) : []
    let syllabicCount = phrasePassings.filter((v) => !v).length

    // Step 4: Duration heuristic — if syllabicCount > sylTokens.length, mark
    // shortest non-passing notes as passing until counts balance
    let adjustedPassings = [...phrasePassings]
    if (syllabicCount > sylTokens.length) {
      const events = getDurationEvents(soprano, doh, phraseStart, phraseEnd)
      // Build array of {index, duration} for non-passing positions, sort ascending duration
      const candidates = events
        .map((e, i) => ({ i, duration: e.duration, passing: adjustedPassings[i] }))
        .filter((c) => !c.passing)
        .sort((a, b) => a.duration - b.duration) // shortest first

      let surplus = syllabicCount - sylTokens.length
      // Convert shortest notes to passing, later indices preferred for ties
      for (const c of candidates) {
        if (surplus <= 0) break
        adjustedPassings[c.i] = true
        surplus--
      }
      syllabicCount = adjustedPassings.filter((v) => !v).length
    }

    // Step 5: Check if still mismatched after heuristic
    if (syllabicCount !== sylTokens.length) {
      warnings.push(
        `abc-melisma: phrase ${phraseIndex} syllable mismatch after heuristic — ` +
        `${syllabicCount} syllabic slots, ${sylTokens.length} syllable tokens. ` +
        `Padding/truncating.`,
      )
    }

    // Step 6: Build token array
    const tokens: string[] = []
    let sylIdx = 0
    for (let ni = 0; ni < noteCount; ni++) {
      if (adjustedPassings[ni]) {
        tokens.push('_')
      } else {
        tokens.push(sylTokens[sylIdx] ?? '*')
        sylIdx++
      }
    }

    return tokens.join(' ')
  }
  ```
