    ```typescript
    /**
     * Pure helpers for detecting and extracting source-authored `w:` lyric lines
     * from an ABC phrase body. Used by NotationRenderer (Phase 04.10) to detect
     * source-authored w: lines and bypass the heuristic w-line construction path
     * (buildWLineFromSolfa / syllabifyForAbc).
     *
     * Pure module — no DOM, no abcjs, no Next/React imports. Safe for SSR and
     * unit tests.
     */

    /**
     * True iff the given phrase body (a string from splitOnPhraseBreaks().phrases[i])
     * contains at least one `w:` line. Used by NotationRenderer to opt into the
     * verified-MusicXML rendering path for Crimond (Phase 04.10) — when present,
     * the heuristic w-line construction (buildWLineFromSolfa / syllabifyForAbc) is
     * skipped and the source-authored lyrics are rendered verbatim.
     */
    export function hasEmbeddedWLines(phraseBody: string): boolean {
      return /^\s*w:/m.test(phraseBody)
    }

    /**
     * Returns the `w:` line(s) in order from a phrase body. Strips the `w: ` prefix.
     * Each returned string is the raw lyric payload as authored in the source ABC.
     */
    export function extractEmbeddedWLines(phraseBody: string): string[] {
      return phraseBody
        .split('\n')
        .map((l) => l.trim())
        .filter((l) => /^w:/.test(l))
        .map((l) => l.replace(/^w:\s*/, ''))
    }
    ```

    Create `src/lib/abc-embedded-lyrics.test.ts` following the PATTERNS.md analog

    in `<behavior>`. Use this exact top-level fixture block:

    ```typescript
    import { describe, it, expect } from 'vitest'
    import { hasEmbeddedWLines, extractEmbeddedWLines } from './abc-embedded-lyrics'

    const PHRASE_WITH_W = `| G2 G G | G F E D |
    w: The Lord's my _ shep- -herd, _ I'll`

    const PHRASE_NO_W = `| G2 G G | G F E D |`

    describe('hasEmbeddedWLines', () => {
      it('returns true when phrase body contains a w: line', () => {
        expect(hasEmbeddedWLines(PHRASE_WITH_W)).toBe(true)
      })
      it('returns false for phrase body with no w: line', () => {
        expect(hasEmbeddedWLines(PHRASE_NO_W)).toBe(false)
      })
      it('tolerates leading whitespace on the w: line', () => {
        expect(hasEmbeddedWLines('| A |\n   w: foo bar')).toBe(true)
      })
      it('returns false on empty string', () => {
        expect(hasEmbeddedWLines('')).toBe(false)
      })
    })

    describe('extractEmbeddedWLines', () => {
      it('strips the w: prefix and returns the payload', () => {
        expect(extractEmbeddedWLines('| A B C |\nw: do re mi')).toEqual(['do re mi'])
      })
      it('preserves multiple w: lines in source order', () => {
        expect(extractEmbeddedWLines('| A |\nw: line one\n| B |\nw: line two'))
          .toEqual(['line one', 'line two'])
      })
      it('returns empty array when no w: line is present', () => {
        expect(extractEmbeddedWLines('| A B C |')).toEqual([])
      })
    })
    ```
