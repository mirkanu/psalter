    MAIN flow (per RESEARCH.md "Conversion script skeleton" + Pitfall handling):
    ```typescript
    async function main() {
      const args = process.argv.slice(2)
      const APPLY = args.includes('--apply')

      // Pitfall #1 mitigation: verify xml2abc is reachable
      // 1. Run xml2abc on the committed fixture (do NOT fetch at runtime)
      const tmpOut = '/tmp/crimond-deboer.abc'
      execSync(`python3 scripts/xml2abc.py ${SOURCE} -o ${tmpOut}`, { stdio: 'inherit' })
      const satb = readFileSync(tmpOut, 'utf-8')

      // 2. Extract soprano
      const soprano = extractMelodyVoice(satb)

      // 3. Parse slur map directly from MusicXML
      const slurMap = parseSlursFromMusicXml(readFileSync(SOURCE, 'utf-8'))
      if (slurMap.length === 0) throw new Error('No notes parsed from MusicXML — check voice filter')

      // 4. Walk slurs+syllables → w-tokens
      const wTokens = buildWTokens(slurMap, PSALM_23_STANZA_1)

      // 5. Insert PHRASE_BREAK markers
      const phrased = insertPhraseBreaks(soprano, 4, PHRASE_BREAK_POSITIONS)

      // 6. Weave w-lines into phrase bodies
      const final = weaveWLinesIntoPhrases(phrased, wTokens)

      // 7. Validate (Pitfall #5 mitigation: also log final length for awareness)
      const split = splitOnPhraseBreaks(final)
      if (split.phrases.length !== 4) {
        throw new Error(`Expected 4 phrases (CM), got ${split.phrases.length}`)
      }
      for (let i = 0; i < split.phrases.length; i++) {
        const heads = countNoteHeads(split.phrases[i])
        const tokens = countWTokens(split.phrases[i])
        if (heads !== tokens) {
          throw new Error(`Phrase ${i}: ${heads} note-heads vs ${tokens} w-tokens — mismatch`)
        }
      }
      console.log(`VALIDATED: 4 phrases, length=${final.length}`)

      // 8. Write fixture (always — even with --apply, for debugging)
      writeFileSync(OUTPUT, final)
      console.log(`Wrote ${OUTPUT}`)

      if (APPLY) {
        if (!process.env.DATABASE_URL) throw new Error('DATABASE_URL not set')
        const pgClient = postgres(process.env.DATABASE_URL)
        const db = drizzle({ client: pgClient, schema })
        const result = await db.update(schema.tunes)
          .set({ abcNotation: final })
          .where(eq(schema.tunes.name, 'Crimond'))
          .returning({ id: schema.tunes.id })
        if (result.length !== 1) {
          await pgClient.end()
          throw new Error(`Expected exactly 1 row updated, got ${result.length}`)
        }
        await pgClient.end()
        console.log(`Applied to tunes.name='Crimond' (id=${result[0].id})`)
      } else {
        console.log('Dry run — re-run with --apply to write to DB')
      }
    }
    main().catch(e => { console.error(e); process.exit(1) })
    ```
