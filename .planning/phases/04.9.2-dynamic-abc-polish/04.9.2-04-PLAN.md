```ts
const halfSize = paginateTuneHalf ? T / 2 : T
const visiblePhraseIndices = paginateTuneHalf
  ? (halfPage === 'A'
      ? Array.from({ length: halfSize }, (_, i) => i)
      : Array.from({ length: halfSize }, (_, i) => i + halfSize))
  : Array.from({ length: T }, (_, i) => i)

// For each visible phrase i, collect w: lines from each visible cycle.
// When tune-half pagination is active, slice each cycle to the matching stanza-half:
//   half-A → first ceil(cycle.length / 2) stanzas (first-of-pair)
//   half-B → remaining stanzas (second-of-pair)
// We re-call mapCycleToPhraseSyllableLines per cycle so grid[i] indexes correctly
// for the active half.
function wLinesForPhrase(i: number): string[] {
  return visibleCycles.flatMap(cycle => {
    const cycleSlice = paginateTuneHalf
      ? (halfPage === 'A'
          ? cycle.slice(0, Math.ceil(cycle.length / 2))
          : cycle.slice(Math.ceil(cycle.length / 2)))
      : cycle
    const grid = mapCycleToPhraseSyllableLines(cycleSlice, tuneMeter, stanzaMeter)
    return grid[i] ?? ['']
  }).filter(s => s.length > 0)
}

visiblePhraseIndices.map(i => (

    key={i}
    abc={buildAbcWithSyllables(buildPhraseAbc(split, i), wLinesForPhrase(i))}
```

```typescript
'use client'

import { useEffect, useState } from 'react'

export function useMediaQuery(query: string): boolean {
  const [matches, setMatches] = useState(false)
  useEffect(() => {
    if (typeof window === 'undefined' || !window.matchMedia) return
    const mql = window.matchMedia(query)
    setMatches(mql.matches)
    const listener = (e: MediaQueryListEvent) => setMatches(e.matches)
    mql.addEventListener('change', listener)
    return () => mql.removeEventListener('change', listener)
  }, [query])
  return matches
}
```
