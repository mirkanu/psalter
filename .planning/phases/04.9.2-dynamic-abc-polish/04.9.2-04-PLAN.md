<data_flow>
For each visible phrase index `i` (per D-16):

```
1. split = splitOnPhraseBreaks(abc)                            // header + phrases[]
2. stanzas = lyrics.split(/\n\s*\n/).map(trim).filter(Boolean) // string[]
3. stanzaPortionsByStanza = stanzas.map(s =>
     splitStanzaIntoPhrasePortions(s, tuneMeter))              // string[][] — for each stanza, an array of N portions
4. // For THIS phrase i, collect the i-th portion from every stanza:
   portionsForPhraseI = stanzaPortionsByStanza.map(portions => portions[i] ?? '')
5. phraseAbc = buildPhraseAbc(split, i)                        // header + phrase body
6. abcWithLyrics = buildAbcWithSyllables(phraseAbc, portionsForPhraseI)
7. <AbcPlayer abc={abcWithLyrics} scale={baseSize / 14} ... />
```

When multiple phrases are visible (standard CM/LM/SM: both phrases on screen at once), the loop runs once per visible phrase, each producing its own AbcPlayer with its own per-phrase w: lines. The user reads phrase 1 + phrase 2 down the page to see each stanza's complete text.
</data_flow>

```tsx
'use client'

import dynamic from 'next/dynamic'
import { Suspense } from 'react'
import { Skeleton } from '@/components/ui/skeleton'
import type { NotationRendererProps } from './NotationRenderer'

/**
 * Client boundary for NotationRenderer. abcjs is DOM-manipulating and must NOT
 * run on the server (project CLAUDE.md rule). This wrapper enforces the rule
 * once so consumer pages can import it directly without configuring dynamic
 * imports themselves.

const NotationRenderer = dynamic(
  () => import('./NotationRenderer').then((m) => m.NotationRenderer),
  { ssr: false }
)

export function NotationRendererClient(props: NotationRendererProps) {
  return (
    <Suspense fallback={<Skeleton className="h-52 w-full rounded-md" />}>
      <NotationRenderer {...props} />
    </Suspense>
  )
}

export type { NotationRendererProps }
```
