```typescript
export interface SplitAbc {
  header: string
  phrases: string[]
}

export function splitOnPhraseBreaks(abc: string): SplitAbc {
  const lines = abc.split('\n')
  const kIdx = lines.findIndex((l) => /^K:/.test(l.trim()))
  if (kIdx === -1) return { header: '', phrases: [abc] }
  const header = lines.slice(0, kIdx + 1).join('\n')
  const body = lines.slice(kIdx + 1).join('\n')
  const phrases = body
    .split(/^\s*%\s*PHRASE_BREAK\s*$/m)
    .map((s) => s.trim())
    .filter(Boolean)
  return { header, phrases: phrases.length ? phrases : [body.trim()] }
}

export function buildPhraseAbc(split: SplitAbc, phraseIndex: number): string {
  if (split.phrases.length === 0) return split.header
  const body = split.phrases[phraseIndex] ?? split.phrases[0]
  return `${split.header}\n${body}`
}
```

```typescript
import { phrasesForMeter } from './abc-phrase-meter-map'

export function splitStanzaIntoPhrasePortions(
  stanzaText: string,
  meter: string | null | undefined,
): string[] {
  const n = phrasesForMeter(meter)
  if (n <= 1) return [stanzaText]
  const lines = stanzaText
    .split('\n')
    .map((l) => l.trim())
    .filter(Boolean)
  if (lines.length === 0) return Array.from({ length: n }, () => '')
  const linesPerPortion = Math.ceil(lines.length / n)
  const out: string[] = []
  for (let i = 0; i < n; i++) {
    const slice = lines.slice(i * linesPerPortion, (i + 1) * linesPerPortion)
    out.push(slice.join('\n'))
  }
  return out
}

export function buildAbcWithSyllables(
  phraseAbc: string,
  stanzaPortionsForThisPhrase: string[],
): string {
  const cleaned = phraseAbc
    .replace(/^w:.*$/gm, '')
    .replace(/\n\n+/g, '\n')
    .trim()
  if (stanzaPortionsForThisPhrase.length === 0) return cleaned
  const wLines = stanzaPortionsForThisPhrase
    .filter((p) => p && p.trim().length > 0)
    .map((p) => `w: ${syllabifyForAbc(p.replace(/\n/g, ' '))}`)
    .join('\n')
  if (!wLines) return cleaned
  return `${cleaned}\n${wLines}`
}
```

```typescript
import { phrasesForMeter } from './abc-phrase-meter-map'
import { splitStanzaIntoPhrasePortions } from './lyrics'

/**
 * Group a flat stanzas[] into stanza-cycles. A cycle = one full pass through
 * the tune. Per D-20:
 *
 *   CM tune × CM psalm        → 1 stanza per cycle (cycleSize = 1)
 *   double-CM tune × CM psalm → 2 stanzas per cycle, paired (cycleSize = 2)
 *
 * Reverse mismatch (stanza requires more phrases than the tune provides) is
 * deferred for Phase 4.9.2 — logs a console.warn and falls back to 1-stanza
 * cycles.
 */
export function groupStanzasIntoCycles(
  stanzas: string[],
  tuneMeter: string | null | undefined,
  stanzaMeter: string | null | undefined,
): string[][] {
  if (stanzas.length === 0) return []
  const T = phrasesForMeter(tuneMeter)
  const S = phrasesForMeter(stanzaMeter)
  if (S > T && T > 0 && S > 0) {
    // eslint-disable-next-line no-console
    console.warn(
      `[stanza-cycles] reverse meter mismatch: tune_phrases=${T} < stanza_phrases=${S}; falling back to 1-stanza cycles`,
    )
    return stanzas.map((s) => [s])
  }
  const cycleSize = Math.max(1, Math.floor(T / Math.max(1, S)))
  const out: string[][] = []
  for (let i = 0; i < stanzas.length; i += cycleSize) {
    out.push(stanzas.slice(i, i + cycleSize))
  }
  return out
}

/**
 * For ONE stanza-cycle, return a T-element array of w: line lists, where
 * element i is the array of w: line strings for phrase index i across this
 * cycle. Per D-19/D-20: each phrase row receives only its slice of each
 * stanza in the cycle.
 */
export function mapCycleToPhraseSyllableLines(
  cycle: string[],
  tuneMeter: string | null | undefined,
  stanzaMeter: string | null | undefined,
): string[][] {
  const T = phrasesForMeter(tuneMeter)
  const S = phrasesForMeter(stanzaMeter)
  const result: string[][] = Array.from({ length: T }, () => [''])
  for (let k = 0; k < cycle.length; k++) {
    const portions = splitStanzaIntoPhrasePortions(cycle[k] ?? '', stanzaMeter)
    for (let p = 0; p < S; p++) {
      const phraseIdx = k * S + p
      if (phraseIdx >= T) break
      result[phraseIdx] = [portions[p] ?? '']
    }
  }
  return result
}
```
