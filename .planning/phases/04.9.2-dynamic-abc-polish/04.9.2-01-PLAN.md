```typescript
/**
 * Number of phrases per stanza based on Scottish Psalter metre conventions.
 * One phrase = one displayed staff line + corresponding lyric line break (D-04).
 *
 * Per D-03: lookup table, NOT algorithmic. Manually verifiable, easy to amend
 * as QA reveals new tunes.
 */
export const PHRASES_PER_STANZA: Record<string, number> = {
  CM: 2,
  LM: 2,
  SM: 2,
  DCM: 4,
  DLM: 4,
  DSM: 4,
  '8.7.8.7': 2,
  '7.6.7.6': 2,
}

/**
 * Normalise a stored tune.meter value to the lookup key, then return the
 * configured phrase count. Unknown / null / undefined yields 1 (whole-tune,
 * no phrase break).
 *
 * Handles forms observed in the DB (STATE.md decision [04-02]):
 *   "CM (common meter, 8.6.8.6)"  -> "CM"
 *   "LM (long meter, 88 88)"      -> "LM"
 *   "DCM"                          -> "DCM"
 *
 * @example
 *   phrasesForMeter("CM (common meter, 8.6.8.6)") // 2
 *   phrasesForMeter(null)                          // 1
 */
export function phrasesForMeter(meter: string | null | undefined): number {
  if (!meter) return 1
  const normalised = meter.trim().split(/\s+/)[0]?.toUpperCase()
  if (!normalised) return 1
  return PHRASES_PER_STANZA[normalised] ?? PHRASES_PER_STANZA[meter.trim()] ?? 1
}
```

```typescript
export interface SplitAbc {
  header: string
  phrases: string[]
}

/**
 * Split an ABC notation string at `% PHRASE_BREAK` comment markers (D-02).
 * The header (everything up to and including the K: line) is captured separately
 * so callers can reconstruct one phrase at a time via buildPhraseAbc().
 *
 * @example
 *   splitOnPhraseBreaks("X:1\nT:Old 100th\nM:L.M.\nK:G\n| G2 G G |\n% PHRASE_BREAK\n| D2 D D |")
 *   // => { header: "X:1\nT:Old 100th\nM:L.M.\nK:G", phrases: ["| G2 G G |", "| D2 D D |"] }
 */
export function splitOnPhraseBreaks(abc: string): SplitAbc {
  const lines = abc.split('\n')
  const kIdx = lines.findIndex((l) => /^K:/.test(l.trim()))
  if (kIdx === -1) return { header: '', phrases: [abc] }
  const header = lines.slice(0, kIdx + 1).join('\n')
  const body = lines.slice(kIdx + 1).join('\n')
  const phrases = body
    .split(/^\s*%\s*PHRASE_BREAK\s*$/m)
    .map((s) => s.trim())
    .filter(Boolean)
  return { header, phrases: phrases.length ? phrases : [body.trim()] }
}

/**
 * Reassemble a single phrase by combining header + the chosen phrase body.
 * Out-of-range indices fall back to phrase 0.
 */
export function buildPhraseAbc(split: SplitAbc, phraseIndex: number): string {
  if (split.phrases.length === 0) return split.header
  const body = split.phrases[phraseIndex] ?? split.phrases[0]
  return `${split.header}\n${body}`
}
```

```typescript
import { phrasesForMeter } from './abc-phrase-meter-map'

/**
 * Split ONE stanza's text into N portions, where N = phrasesForMeter(meter).
 * Each portion contains only the lyric lines that fall under that phrase's
 * melody, per D-16.
 *
 * Mapping rule: with L non-blank lines and N phrases, each portion gets
 * Math.ceil(L / N) consecutive lines (final portion may have fewer).
 *
 *   CM   (N=2)  with 4 lyric lines  ->  [lines 1+2, lines 3+4]
 *   DCM  (N=4)  with 8 lyric lines  ->  [1+2, 3+4, 5+6, 7+8]
 *   DCM  (N=4)  with 4 lyric lines  ->  [1, 2, 3, 4]
 *
 * Verse-number prefix (e.g. leading "1" in "1The Lord's my Shepherd") is
 * preserved on portion[0] only.
 *
 * @example
 *   splitStanzaIntoPhrasePortions(
 *     "1The Lord's my Shepherd\nI'll not want\nHe makes me down to lie\nIn pastures green",
 *     "CM"
 *   )
 *   // => ["1The Lord's my Shepherd\nI'll not want", "He makes me down to lie\nIn pastures green"]
 */
export function splitStanzaIntoPhrasePortions(
  stanzaText: string,
  meter: string | null | undefined,
): string[] {
  const n = phrasesForMeter(meter)
  if (n <= 1) return [stanzaText]
  const lines = stanzaText
    .split('\n')
    .map((l) => l.trim())
    .filter(Boolean)
  if (lines.length === 0) return Array.from({ length: n }, () => '')
  const linesPerPortion = Math.ceil(lines.length / n)
  const out: string[] = []
  for (let i = 0; i < n; i++) {
    const slice = lines.slice(i * linesPerPortion, (i + 1) * linesPerPortion)
    out.push(slice.join('\n'))
  }
  return out
}

/**
 * Inject syllabified per-stanza portions as w: lines into a SINGLE phrase's ABC.
 *
 * Per D-16: each phrase shows only its corresponding lyric portion of each
 * stanza, NOT the full stanza. The caller is responsible for computing each
 * stanza's portion for THIS phrase (typically via splitStanzaIntoPhrasePortions).
 *
 * - Strips any pre-existing w: lines (defensive)
 * - Collapses blank lines (abcjs treats `\n\n` as new tune — Pitfall 1)
 * - Each non-empty portion becomes ONE w: line; multiple w: lines render as
 *   stacked verses below this phrase's staff
 * - Each portion is run through syllabifyForAbc to produce hyphen-suffix tokens
 * - Empty-string portions are skipped (no w: line for that stanza on this phrase)
 *
 * Per D-12 (lyrics logic in one place) and D-13 (current phrase only).
 *
 * @example
 *   // CM tune, phrase 1, 3 stanzas — each portion = lines 1+2 of its stanza
 *   buildAbcWithSyllables(
 *     "X:1\nK:G\n|G2 G G|",
 *     [
 *       "1The Lord's my Shepherd\nI'll not want",   // stanza 1, phrase-1 portion
 *       "2When in the way of sin\nthat man doth go", // stanza 2, phrase-1 portion
 *       "3God is our refuge and our strength\nin straits a present aid", // stanza 3, phrase-1 portion
 *     ]
 *   )
 *   // => ABC with three w: lines stacked, one per stanza, each showing only
 *   //    that stanza's first two lyric lines (the part that sings under phrase 1)
 */
export function buildAbcWithSyllables(
  phraseAbc: string,
  stanzaPortionsForThisPhrase: string[],
): string {
  const cleaned = phraseAbc
    .replace(/^w:.*$/gm, '')
    .replace(/\n\n+/g, '\n')
    .trim()
  if (stanzaPortionsForThisPhrase.length === 0) return cleaned
  const wLines = stanzaPortionsForThisPhrase
    .filter((p) => p && p.trim().length > 0)
    .map((p) => `w: ${syllabifyForAbc(p.replace(/\n/g, ' '))}`)
    .join('\n')
  if (!wLines) return cleaned
  return `${cleaned}\n${wLines}`
}
```
