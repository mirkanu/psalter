into `targetStaffwidth = (containerWidth * staffWidthFactor) / effectiveScale`. So a fixed
`INLINE_STAFF_NOTATION_SCALE` must MATCH the current default inline-Staff density (mobile
chromeless default baseSize = 13 → 13/14 ≈ 0.929) to avoid a visible size jump on first
deploy. Use `13 / 14` as the constant; abcjs `responsive:'resize'` scales the SVG viewBox to
fill the container width regardless, so the notation still fills width on all viewports.

```ts
export const SPLIT_LEAF_NOTATION_SCALE = 1
// Fixed constant mirroring SPLIT_LEAF_NOTATION_SCALE (locked decision 2, UI-SPEC §b).
// 13/14 preserves the CURRENT inline-Staff density: mobile chromeless default
// baseSize was 13, previously scaled as baseSize/14. Decoupled from the lyric-only
// A+/A− control so the staff no longer grows/shrinks when the user zooms lyrics.
export const INLINE_STAFF_NOTATION_SCALE = 13 / 14

export function computeNotationScale(opts: {
  viewMode: string
  chromeless: boolean
  baseSize: number
}): number {
  const isSplit = opts.viewMode === 'staff-split' || opts.viewMode === 'solfege-split'
  if (isSplit) return SPLIT_LEAF_NOTATION_SCALE
  // Decouple ONLY the chromeless (mobile singing) inline Staff view. Non-chromeless
  // desktop /tunes/[id] staff keeps the coupled baseSize/14 behaviour (out of scope).
  if (opts.chromeless && opts.viewMode === 'staff') return INLINE_STAFF_NOTATION_SCALE
  return opts.baseSize / 14
}
```
