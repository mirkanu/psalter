```typescript
import { test, expect } from '@playwright/test';

test.describe('PlayMiniBar Positioning', () => {
  test('renders above GlassBottomBar, no scroll needed', async ({ page }) => {
    await page.setViewportSize({ width: 375, height: 667 });
    await page.goto('/psalms/23');
    
    const playBar = page.locator('[data-play-mini-bar]');
    const bottomBar = page.locator('[data-glass-bottom-bar]');
    
    // Both should be visible without scrolling
    await expect(playBar).toBeVisible();
    await expect(bottomBar).toBeVisible();
    
    // PlayMiniBar should be above GlassBottomBar in viewport
    const playBarBox = await playBar.boundingBox();
    const bottomBarBox = await bottomBar.boundingBox();
    expect(playBarBox.y + playBarBox.height).toBeLessThanOrEqual(bottomBarBox.y + 5); // 5px tolerance
  });

  test('desktop: floating card above bottom bar', async ({ page }) => {
    await page.setViewportSize({ width: 1200, height: 800 });
    await page.goto('/psalms/23');
    
    const playBar = page.locator('[data-play-mini-bar]');
    await expect(playBar).toHaveClass(/fixed/);
    await expect(playBar).toHaveClass(/bottom-\[56px\]/);
    await expect(playBar).toHaveClass(/right-4/);
  });

  test('auto-hides on scroll down, shows on scroll up', async ({ page }) => {
    await page.setViewportSize({ width: 375, height: 667 });
    await page.goto('/psalms/23');
    
    const playBar = page.locator('[data-play-mini-bar]');
    await expect(playBar).toBeVisible();
    
    // Scroll down
    await page.mouse.wheel(0, 300);
    await page.waitForTimeout(300); // wait for debounce
    await expect(playBar).toBeHidden(); // or have translate-y-full class
    
    // Scroll up
    await page.mouse.wheel(0, -100);
    await page.waitForTimeout(300);
    await expect(playBar).toBeVisible();
  });

  test('collapse button works independently of auto-hide', async ({ page }) => {
    await page.setViewportSize({ width: 375, height: 667 });
    await page.goto('/psalms/23');
    
    const playBar = page.locator('[data-play-mini-bar]');
    const collapseBtn = playBar.locator('button[aria-label="Collapse player"]');
    
    await collapseBtn.click();
    await expect(playBar).toBeHidden();
    
    // Scroll up should NOT restore if manually collapsed
    await page.mouse.wheel(0, -200);
    await page.waitForTimeout(300);
    await expect(playBar).toBeHidden();
  });
});
```
