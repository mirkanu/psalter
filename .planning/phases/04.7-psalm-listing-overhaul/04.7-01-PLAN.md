**File 1: /tmp/test-plr-01.js** — PLR-01 dense grid
Tests:
1. Navigate to `https://psalter.gsdlabs.dev/psalms`
2. Wait for page to reach `networkidle`
3. Count elements matching CSS selector `[data-psalm-box]` — assert count >= 150
4. Assert at least one element with `data-psalm-box` attribute containing text "23" is visible
5. Log PASS or FAIL with count

```javascript
const { chromium } = require('/usr/lib/node_modules/playwright');

(async () => {
  const browser = await chromium.launch({
    executablePath: '/tmp/pw-browsers/chromium-1217/chrome-linux/chrome',
    args: ['--no-sandbox']
  });
  const page = await browser.newPage();
  try {
    await page.goto('https://psalter.gsdlabs.dev/psalms', { waitUntil: 'networkidle', timeout: 30000 });
    const boxes = await page.locator('[data-psalm-box]').count();
    console.log(`PLR-01: psalm box count = ${boxes}`);
    if (boxes < 150) throw new Error(`Expected >= 150 boxes, got ${boxes}`);
    console.log('PLR-01: PASS');
  } catch (e) {
    console.error('PLR-01: FAIL', e.message);
    process.exit(1);
  } finally {
    await browser.close();
  }
})();
```

**File 2: /tmp/test-plr-02.js** — PLR-02 search (number + text + Enter)
Tests:
1. Navigate to `/psalms`, wait networkidle
2. Fill search input (aria-label="Search psalms") with "23"
3. Wait 300ms for client-side filter to settle
4. Assert the first `[data-psalm-box]` in the DOM has text content containing "23"
5. Clear search, type "Lord"
6. Wait 300ms
7. Assert at least one `[data-psalm-box]` is visible
8. Assert at least one `<strong>` inside a `[data-psalm-box]` contains text matching /lord/i (bold snippet present)
9. Press Enter on the search input
10. Wait for navigation — assert URL changes to `/psalms/` followed by a number

```javascript
const { chromium } = require('/usr/lib/node_modules/playwright');

(async () => {
  const browser = await chromium.launch({
    executablePath: '/tmp/pw-browsers/chromium-1217/chrome-linux/chrome',
    args: ['--no-sandbox']
  });
  const page = await browser.newPage();
  try {
    await page.goto('https://psalter.gsdlabs.dev/psalms', { waitUntil: 'networkidle', timeout: 30000 });

    // Number mode
    await page.fill('[aria-label="Search psalms"]', '23');
    await page.waitForTimeout(400);
    const firstBox = page.locator('[data-psalm-box]').first();
    const firstText = await firstBox.textContent();
    if (!firstText?.includes('23')) throw new Error(`Expected first box to contain "23", got: ${firstText}`);
    console.log('PLR-02a (number sort): PASS');

    // Text mode with bold snippet
    await page.fill('[aria-label="Search psalms"]', '');
    await page.fill('[aria-label="Search psalms"]', 'Lord');
    await page.waitForTimeout(400);
    const boldCount = await page.locator('[data-psalm-box] strong').count();
    if (boldCount === 0) throw new Error('Expected at least one <strong> bold snippet for "Lord" search');
    console.log(`PLR-02b (bold snippet): PASS — ${boldCount} bold matches`);

    // Enter key navigation
    await page.keyboard.press('Enter');
    await page.waitForURL(/\/psalms\/\d+/, { timeout: 5000 });
    console.log('PLR-02c (Enter navigate): PASS — URL:', page.url());

    console.log('PLR-02: PASS');
  } catch (e) {
    console.error('PLR-02: FAIL', e.message);
    process.exit(1);
  } finally {
    await browser.close();
  }
})();
```

```javascript
const { chromium } = require('/usr/lib/node_modules/playwright');

(async () => {
  const browser = await chromium.launch({
    executablePath: '/tmp/pw-browsers/chromium-1217/chrome-linux/chrome',
    args: ['--no-sandbox']
  });
  const page = await browser.newPage();
  try {
    await page.goto('https://psalter.gsdlabs.dev/psalms', { waitUntil: 'networkidle', timeout: 30000 });

    // Panel is collapsed by default
    const panelBefore = await page.locator('#advanced-panel').count();
    if (panelBefore !== 0) throw new Error('Advanced panel should be hidden by default');
    console.log('PLR-03a (collapsed by default): PASS');

    // Click to open
    await page.click('button:has-text("Advanced Filters")');
    await page.waitForTimeout(300);
    await page.locator('#advanced-panel').waitFor({ state: 'visible', timeout: 3000 });
    console.log('PLR-03b (panel opens): PASS');

    // Checkboxes present
    const cb1 = await page.locator('#show-first-line').count();
    const cb2 = await page.locator('#show-meter').count();
    if (cb1 === 0) throw new Error('Missing #show-first-line checkbox');
    if (cb2 === 0) throw new Error('Missing #show-meter checkbox');
    console.log('PLR-03c (checkboxes present): PASS');

    // Meter select present
    const meterSelect = await page.locator('[aria-label="Filter by meter"]').count();
    if (meterSelect === 0) throw new Error('Missing meter filter select');
    console.log('PLR-03d (meter select present): PASS');

    // Click to close
    await page.click('button:has-text("Advanced Filters")');
    await page.waitForTimeout(300);
    const panelAfter = await page.locator('#advanced-panel').count();
    if (panelAfter !== 0) throw new Error('Advanced panel should collapse after second click');
    console.log('PLR-03e (panel collapses): PASS');

    console.log('PLR-03: PASS');
  } catch (e) {
    console.error('PLR-03: FAIL', e.message);
    process.exit(1);
  } finally {
    await browser.close();
  }
})();
```
