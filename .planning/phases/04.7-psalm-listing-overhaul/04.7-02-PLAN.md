    .select({

function renderSnippet(snippet: string, query: string) {
  if (!query) return <span>{snippet}</span>
  const escaped = query.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
  const parts = snippet.split(new RegExp(`(${escaped})`, 'gi'))
  return (
    <>
      {parts.map((part, i) =>
        part.toLowerCase() === query.toLowerCase()
          ? <b key={i}>{part}</b>
          : <span key={i}>{part}</span>
      )}
    </>
  )
}

```typescript
function buildSnippet(text: string | null, query: string, maxLen = 80): string | null {
  if (!text || !query) return null
  const lower = text.toLowerCase()
  const lowerQuery = query.toLowerCase()
  const idx = lower.indexOf(lowerQuery)
  if (idx === -1) return null
  const start = Math.max(0, idx - 20)
  const end = Math.min(text.length, start + maxLen)
  let excerpt = text.slice(start, end)
  if (start > 0) excerpt = '…' + excerpt
  if (end < text.length) excerpt = excerpt + '…'
  return excerpt
}
```

**filteredPsalms with full search logic:**
```typescript
const filteredPsalms = useMemo(() => {
  // Apply meter filter first (applies in all modes)
  const meterFiltered = meterFilter === 'all'
    ? psalms
    : psalms.filter((p) => p.meter === meterFilter)

  if (!trimmedQuery) {
    // No search — return meter-filtered list in psalm-number order
    return meterFiltered.map((p) => ({ ...p, snippet: null as string | null }))
  }

  if (isNumeric) {
    // Number mode: sort all meter-filtered psalms by numeric distance
    const target = parseInt(trimmedQuery, 10)
    return [...meterFiltered]
      .sort((a, b) => Math.abs(a.id - target) - Math.abs(b.id - target))
      .map((p) => ({ ...p, snippet: null as string | null }))
  }

  // Text mode: filter + inject snippet + sort by relevance
  const lowerQuery = trimmedQuery.toLowerCase()
  const withMatch = meterFiltered
    .map((p) => {
      const firstLineSnippet = buildSnippet(p.firstLine, trimmedQuery)
      const kjvSnippet = buildSnippet(p.kjvExcerpt, trimmedQuery)
      const snippet = firstLineSnippet ?? kjvSnippet
      // Relevance: exact word in firstLine = 0, partial firstLine match = 1, kjv match = 2
      const fl = p.firstLine?.toLowerCase() ?? ''
      const relevance = fl.split(/\s+/).includes(lowerQuery) ? 0
        : fl.includes(lowerQuery) ? 1
        : 2
      return { ...p, snippet, relevance }
    })
    .filter((p) => p.snippet !== null)
    .sort((a, b) => a.relevance - b.relevance)

  return withMatch.map(({ relevance: _r, ...p }) => p)
}, [psalms, trimmedQuery, isNumeric, meterFilter])

const topResult = filteredPsalms[0] ?? null

import { db } from "@/db"
import { psalms, psalmVersions } from "@/db/schema"
import { eq, asc, sql } from "drizzle-orm"
import { PsalmListingGrid } from "@/components/PsalmListingGrid"
import type { Metadata } from "next"

export const metadata: Metadata = {
  title: "Psalms | CPRC Psalter",
  description: "Browse all 150 psalms of the Scottish Psalter.",
}

export default async function PsalmsPage() {
  const rows = await db
    .select({
      id: psalms.id,
      firstLine: psalmVersions.firstLine,
      meter: psalmVersions.meter,
      kjvExcerpt: sql<string>`LEFT(${psalms.kjvText}, 120)`.as('kjv_excerpt'),
    })
    .from(psalms)
    .leftJoin(psalmVersions, eq(psalmVersions.psalmId, psalms.id))
    .orderBy(asc(psalms.id))

  // Deduplicate: keep only the FIRST version row per psalm
  const seen = new Set<number>()
  const uniqueRows = rows.filter((r) => {
    if (seen.has(r.id)) return false
    seen.add(r.id)
    return true
  })

  return (
    <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-6 md:py-8">
      <div className="mb-6 mt-4">
        <h1 className="font-sans text-3xl md:text-4xl font-bold text-foreground mb-3">
          Psalms
        </h1>
        <p className="text-muted-foreground text-base">
          Browse all 150 psalms of the Scottish Psalter.
        </p>
      </div>
      <PsalmListingGrid psalms={uniqueRows} />
    </div>
  )
}
```
