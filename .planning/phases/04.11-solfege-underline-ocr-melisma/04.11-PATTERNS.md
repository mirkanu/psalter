  const rawFile = fs.readFileSync(imagePath)
  return rawFile.length > 3.5 * 1024 * 1024

**`callClaude` retry/backoff — REUSE verbatim** (lines 99-132):
```typescript
async function callClaude(imageBuffers: Buffer | Buffer[], prompt: string, maxTokens = 3000): Promise<string> {
  const buffers = Array.isArray(imageBuffers) ? imageBuffers : [imageBuffers]
  const imageContent = buffers.map(buf => ({
    type: 'image' as const,
    source: { type: 'base64' as const, media_type: 'image/jpeg' as const, data: buf.toString('base64') },
  }))
  const MAX_RETRIES = 4
  let lastError: unknown
  for (let attempt = 0; attempt <= MAX_RETRIES; attempt++) {
    try {
      const response = await client.messages.create({
        model: 'claude-haiku-4-5-20251001',
        max_tokens: maxTokens,
        messages: [{ role: 'user', content: [...imageContent, { type: 'text' as const, text: prompt }] }],
      })
      return response.content[0].type === 'text' ? response.content[0].text : ''
    } catch (err) {
      lastError = err
      const isOverloaded = err instanceof Error && (
        err.message.includes('overloaded_error') ||
        err.message.includes('529') ||
        (err as { status?: number }).status === 529
      )
      if (!isOverloaded || attempt === MAX_RETRIES) throw err
      await new Promise(r => setTimeout(r, 5000 * 2 ** attempt))
    }
  }
  throw lastError
}
```
**Action for v3:** export `callClaude` from v2 so v3 can re-use it directly. Call with `maxTokens = 4000` (structured JSON is larger than the v2 string).

**Token-array → boolean[] mapping shape** (`solfege-parser.ts` lines 443-453, `getPassingPositions`):
```typescript
export function getPassingPositions(
  soprano: string,
  doh: string,
  _time: string,
): boolean[] {
  const warnings: string[] = []
  const tonic = DOH_SEMITONES[doh] ?? 0
  const { sharps, flats } = keyAccidentalsForDoh(doh)
  const events = parseVoiceLine(soprano, tonic, warnings, sharps, flats)
  return events.map((e) => e.passing === true)
}
```
**Action for `build-embedded-wline.ts`:** the inverse / replacement — instead of producing `boolean[]` from solfège text + dot-pair heuristic, it consumes Vision's `underlinedFlags[]` directly and the stanza-1 syllables, walks them per RESEARCH §Pattern 2 (`syllableIdx++` on non-underlined, emit `_` on underlined), and returns the ABC body with embedded `w:` lines.

**Phrase-splitting reuse** (`src/lib/abc-phrases.ts` exports — DO NOT hand-roll):
```typescript
export function splitOnPhraseBreaks(abc: string): SplitAbc
export function buildPhraseAbc(split: SplitAbc, phraseIndex: number): string
export function countNoteHeads(abc: string): number
```
**Action:** import and use `splitOnPhraseBreaks` to slice the existing `abc_notation`, then for each phrase body compute `countNoteHeads(body)` and assert equality with the matching slice of underlined+non-underlined tokens (the single-source-of-truth invariant from RESEARCH Pattern 4).

**Validation-fail shape (return discriminated union)** — pattern from `solfege-parser.ts` `solFaToAbc` return:
```typescript
return { abc, bars, warnings }
```
**Action:** return `{ abc: string; warnings: string[]; passesValidation: boolean; perPhrase: PerPhraseDiagnostics[] }` so the caller (`ocr-melisma-batch.ts`) can decide whether to skip the DB write (D-05 heuristic fallback).
