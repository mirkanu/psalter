```typescript
/**
 * Pure module — no DOM, no abcjs, no Next/React imports, no DB, no fs, no Vision.
 *
 * buildEmbeddedWline: deterministic transform from
 *   (Vision tokens + per-token underlined flags + stanza-1 syllables + existing ABC body)
 * to an ABC body with embedded w: lines containing _ continuation tokens at every
 * underlined note position.
 *
 * Implements lyric-to-note-alignment.md §6 algorithm on solfège-OCR output (Phase 04.11 D-01).
 * Single source of truth for syllable→note mapping; PHRASE_BREAK markers are visual-only.
 */
import { splitOnPhraseBreaks, countNoteHeads } from './abc-phrases'

export interface BuildEmbeddedWlineInput {
  tokens: string[]
  underlined: boolean[]
  syllables: string[]
  existingAbc: string
}
export interface BuildEmbeddedWlineResult {
  abc: string
  passesValidation: boolean
  warnings: string[]
  perPhrase: Array&lt;{ phraseIdx: number; noteHeadCount: number; wTokenCount: number }&gt;
}
export function buildEmbeddedWline(input: BuildEmbeddedWlineInput): BuildEmbeddedWlineResult { ... }
```

Algorithm steps (implement exactly):
1. Validate input invariants up front: `tokens.length === underlined.length`. If not, return `{ abc: input.existingAbc, passesValidation: false, warnings: ['tokens/underlined length mismatch'], perPhrase: [] }`.
2. Build the full `w:` token stream from (tokens, underlined, syllables):
   - `syllIdx = 0`; for each i in tokens: if `underlined[i]` push `'_'`; else push `syllables[syllIdx++]` (or `'*'` if syllIdx out of bounds — and record warning).
3. Use `splitOnPhraseBreaks(input.existingAbc)` to get phrase bodies and separators (read the file to confirm exact return shape; this code must use the actual export).
4. For each phrase body, `n = countNoteHeads(body)`; slice the next `n` tokens from the w-stream; emit `w: ` + tokens.join(' '); record `perPhrase` entry with both counts.
5. If `n` exhausts the stream before all phrases get tokens, OR the stream has leftover tokens after the last phrase, record warnings and set `passesValidation = false`.
6. Rebuild the ABC by interleaving phrase bodies + their w-lines + separators. PHRASE_BREAK separators preserved verbatim.
7. Final pass: assert `perPhrase.every(p =&gt; p.noteHeadCount === p.wTokenCount)`; folder result into `passesValidation`.
