async function resolveStanza1Syllables(tuneId: number, db: ...): Promise<string[] | null> {
  // schemaPath from .planning/phases/04.11-.../stanza1-probe.json (Task 1.5):
  //   schema.tunes → schema.psalmVersionTunes.tuneId → schema.psalmVersions.lyricsStructured
  //     → cycles[0].stanzas[0].(lines[0]|metricalLines[0]).syllables
  const pvts = await db.select().from(schema.psalmVersionTunes).where(eq(schema.psalmVersionTunes.tuneId, tuneId))
  if (pvts.length === 0) return null
  const pv = await db.select({ ls: schema.psalmVersions.lyricsStructured }).from(schema.psalmVersions).where(eq(schema.psalmVersions.id, pvts[0].psalmVersionId))
  if (pv.length !== 1) return null
  const ls = pv[0].ls as any
  const stanza1 = ls?.cycles?.[0]?.stanzas?.[0]
  const firstLine = stanza1?.lines?.[0] ?? stanza1?.metricalLines?.[0]
  const syllables = firstLine?.syllables
  return Array.isArray(syllables) ? syllables : null
}
