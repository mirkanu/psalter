3. Add two test blocks:

**Test 1: SVG rendering check (automated)**
```typescript
for (const [tuneName, slug, meter] of SPOT_CHECK_TUNES) {
  test(`tune page renders abcjs SVG notation: ${tuneName} (${meter})`, async ({ page }) => {
    await page.goto(`/tunes/${slug}`, { waitUntil: 'networkidle' })
    
    // Page should not show an error
    await expect(page).not.toHaveTitle(/Error|404|Not Found/)
    
    // The tune name heading should be visible
    await expect(page.locator('h1')).toContainText(tuneName, { ignoreCase: true })
    
    // abcjs renders SVG elements — confirm at least one SVG is present with content
    const svg = page.locator('svg').first()
    await expect(svg).toBeVisible({ timeout: 5000 })
    
    // SVG should contain path elements (drawn notes — empty SVG has no paths)
    const paths = page.locator('svg path')
    const pathCount = await paths.count()
    expect(pathCount).toBeGreaterThan(0)
  })
}
```

**Test 2: Staff/Solfège toggle check (for tunes that have solfège images)**
```typescript
test('Staff/Solfège toggle switches between abcjs SVG and solfège JPG', async ({ page }) => {
  // Use a tune known to have both abc_notation and a solfège image
  // Dundee always has the solfège image on disk
  await page.goto('/tunes/dundee', { waitUntil: 'networkidle' })
  
  // Staff view (default) shows SVG
  const svg = page.locator('svg').first()
  await expect(svg).toBeVisible({ timeout: 5000 })
  
  // Click the Solfège toggle button
  const solfegeBtn = page.getByRole('button', { name: /solfège|solfege/i })
  if (await solfegeBtn.count() > 0) {
    await solfegeBtn.click()
    // After toggle, SVG should be hidden and a JPG/img should appear
    await expect(svg).not.toBeVisible()
    const solfegeImg = page.locator('img[src*="solfege"]')
    await expect(solfegeImg).toBeVisible({ timeout: 3000 })
    
    // Toggle back to Staff
    const staffBtn = page.getByRole('button', { name: /staff/i })
    await staffBtn.click()
    await expect(svg).toBeVisible({ timeout: 3000 })
  }
})
```
