<objective>
Fix the Contemplation sharps-as-naturals audio bug by appending a trailing `|`
to every sub-line emitted by `splitMusicIntoSubLines` in
`src/components/notation/NotationRenderer.tsx`.

Root cause is documented in
`.planning/debug/contemplation-sharps-as-naturals.md`:
abcjs 6.6.3 synth scopes explicit accidentals (`^`, `_`, `=`) to the next `|`
barline. Sub-lines emitted without a trailing bar leak the final note's
accidental across the `w:` lyric line into the next phrase's leading notes.
In K:Eb this manifests as a phrase-ending `=e` (E natural) bleeding into the
next phrase's `e`, making written Eb sound as E natural (heard as "sharp
playing as natural").

The fix is a one-liner on the return of `splitMusicIntoSubLines`. The visual
SVG is unaffected (abcjs already gives each music line its own staff system
with implicit visual accidental reset).

Purpose: correct audio playback on Contemplation and any other tunes whose
phrase bodies don't end in a bar AND carry explicit accidentals that recur
in the next phrase.

Output: one modified file, one unit test, Playwright + MIDI verification.
</objective>

function splitMusicIntoSubLines(body: string, n: number): string[] {
  if (n <= 1) return [body]
  const segs = body.split(/(\|)/).filter((s) => s.length > 0)
  const measures: string[] = []
  let acc = ''
  for (const s of segs) {
    acc += s
    if (s === '|') { measures.push(acc.trim()); acc = '' }
  }
  if (acc.trim()) measures.push(acc.trim())
  const realMeasures = measures
    .filter((m) => m && m !== '|')
    .filter((m) => !/^\s*[zxZ]\d*\s*$/.test(m))
  if (realMeasures.length < 2) return [body]
  const per = Math.max(1, Math.ceil(realMeasures.length / n))
  const lines: string[] = []
  for (let k = 0; k < realMeasures.length; k += per) {
    lines.push(realMeasures.slice(k, k + per).join(' '))
  }
  return lines  // <-- FIX TARGET: ensure each line ends with `|`
}
